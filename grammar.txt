(* ── Root ───────────────────────────────────── *)
Root         ::= ".{" GroupList "}"         (* object root *)
               | ".[" GroupList "]"         (* array  root *)

(* ── Groups ─────────────────────────────────── *)
(*  GroupSep allows both comma (inline) and      *)
(*  newlines (block) as equivalent separators.   *)
GroupList    ::= Group (GroupSep Group)*
GroupSep     ::= "," | NL+

Group        ::= LeafGroup | BlockGroup

(*  Inline: type tag followed by space-separated keys.  *)
(*  Terminated by GroupSep or closing delimiter.        *)
LeafGroup    ::= TypeExpr Identifier+

(*  Block: type tag opens a paren scope.                *)
(*  Each line inside is one key with an optional shape. *)
BlockGroup   ::= TypeExpr "(" NL+ KeyLine+ ")"

(* ── Keys ───────────────────────────────────── *)
KeyLine      ::= Identifier "?"? RecordBody? NL+

(*  Inline record: everything fits on one line.  *)
(*  Block record:  nested shape spans lines.     *)
RecordBody   ::= "{" GroupList "}"
               | "{" NL+ GroupList NL+ "}"

(* ── Type Expression ─────────────────────────── *)
TypeExpr     ::= AtomicType ("|" AtomicType)*
AtomicType   ::= "A."* BaseType "?"?
BaseType     ::= "S" | "N" | "B" | "L" | "X" | "O"

(* ── Lexical ─────────────────────────────────── *)
Identifier   ::= [a-zA-Z_][a-zA-Z0-9_]*
NL           ::= "\n" | "\r\n"
