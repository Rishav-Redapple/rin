{
  function type(name, arrays, nullable) {
    return { kind: "type", name: name, arrays: arrays, nullable: nullable };
  }

  function key(name, optional, body) {
    var result = { name: name, optional: optional };
    if (body !== null) result.body = body;
    return result;
  }

  function group(typeExpression, keys) {
    return { kind: "group", type: typeExpression, keys: keys };
  }

  function objectGroup(name, arrays, optional, body) {
    return group(type("Object", arrays, false), [key(name, optional, body)]);
  }
}

// ================================================================
// RIN v3 PEG grammar
//
// A root is a record enclosed by { }.
// Groups are separated by a newline, comma, or semicolon.
// A newline is significant between groups; spaces alone separate a
// type from its keys and keys from one another.
// ================================================================

Start
  = lead:Trivia root:Root trail:Trivia EOF { return root; }

Root
  = "{" body:RecordBodyContent? Trivia "}" {
      return { kind: "root", type: "Object", groups: body !== null ? body : [] };
    }
  / name:Identifier Trivia "{" body:RecordBodyContent? Trivia "}" {
      return { kind: "root", name: name, type: "Object", groups: body !== null ? body : [] };
    }
  / "[" Trivia "{" body:RecordBodyContent? Trivia "}" Trivia "]" {
      return { kind: "root", type: "Array<Object>", groups: body !== null ? body : [] };
    }

RecordBody
  = "{" body:RecordBodyContent? Trivia "}" { return body !== null ? body : []; }

RecordBodyContent
  = lead:Trivia first:Group rest:(Separator Trivia group:Group { return group; })* {
      return [first].concat(rest.map(function (item) { return item; }));
    }

// Explicit primitive group, e.g.:
//   String name role address
//   Number id ref_id?
//   String[] ruleset
TypedGroup
  = expression:TypeExpression required:RequiredSpace keys:KeyList {
      return group(expression, keys);
    }

// Object-key shorthand, e.g.:
//   tags[]{String name; Number id}
//   profile{String name}
// Object-key names cannot be primitive names. This makes the first token
// unambiguous: it is either a primitive type or a non-primitive object key.
ObjectGroup
  = name:NonPrimitiveIdentifier arrays:ArraySuffix* optional:OptionalMarker? body:RecordBody {
      return objectGroup(name, arrays.length, optional !== null, body);
    }

Group
  = typed:TypedGroup { return typed; }
  / object:ObjectGroup { return object; }

KeyList
  = first:Key rest:(RequiredSpace key:Key { return key; })* {
      return [first].concat(rest.map(function (item) { return item; }));
    }

Key
  = name:Identifier optional:OptionalMarker? {
      return key(name, optional !== null, null);
    }

TypeExpression
  = first:AtomicType rest:(UnionOperator type:AtomicType { return type; })* {
      var members = [first].concat(rest.map(function (item) { return item; }));
      return members.length === 1 ? first : { kind: "union", members: members };
    }

AtomicType
  = name:Primitive arrays:ArraySuffix* {
      return type(name, arrays.length, false);
    }

Primitive
  = "String" !IdentifierPart { return "String"; }
  / "Number" !IdentifierPart { return "Number"; }
  / "Boolean" !IdentifierPart { return "Boolean"; }
  / "Null" !IdentifierPart { return "Null"; }
  / "Any" !IdentifierPart { return "Any"; }

NonPrimitiveIdentifier
  = !Primitive Identifier

ArraySuffix
  = "[]"

UnionOperator
  = Trivia "|" Trivia

OptionalMarker
  = "?"

Separator
  = Trivia "," Trivia
  / Trivia ";" Trivia
  / LineBreak+

RequiredSpace
  = [ \t]+ Comment?

Trivia
  = (Horizontal / CommentLine / LineBreak)*

Horizontal
  = [ \t\r]+

CommentLine
  = [ \t]* Comment LineBreak

Comment
  = "#" [^\r\n]*

LineBreak
  = [ \t]* ("\r\n" / "\n")

Identifier
  = head:IdentifierStart tail:IdentifierPart* {
      return head + tail.join("");
    }

IdentifierStart
  = [a-zA-Z_]

IdentifierPart
  = [a-zA-Z0-9_]

EOF
  = !.
